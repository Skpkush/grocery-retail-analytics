## Headline KPIs by quarter, 2017.

`sql/analysis/01_kpi_by_quarter.sql`

```sql
-- Headline KPIs by quarter, 2017.
-- Business question: "How is the business trending through the year?"
SELECT
  d.quarter_name AS quarter,
  ROUND(SUM(s.sales_value), 2) AS net_sales,
  COUNT(DISTINCT s.basket_id) AS baskets,
  COUNT(DISTINCT s.household_id) AS households,
  ROUND(SUM(s.sales_value) / NULLIF(COUNT(DISTINCT s.basket_id), 0), 2) AS avg_basket_value,
  ROUND(SUM(s.units) / NULLIF(COUNT(DISTINCT s.basket_id), 0), 2) AS units_per_basket,
  ROUND(100.0 * SUM(s.retail_disc + s.coupon_match_disc) / NULLIF(SUM(s.gross_sales), 0), 1)
    AS discount_rate_pct,
  ROUND(100.0 * SUM(CASE WHEN p.brand = 'Private' THEN s.sales_value END)
        / NULLIF(SUM(s.sales_value), 0), 1) AS private_label_share_pct
FROM fct_sales s
JOIN dim_date d ON d.date_key = s.business_date
JOIN dim_product p ON p.product_id = s.product_id
WHERE s.is_merch
GROUP BY d.quarter_number, d.quarter_name
ORDER BY d.quarter_number;
```

## Coupon incrementality proxy: was the redeeming household already buying the product?

`sql/analysis/10_coupon_incrementality.sql`

```sql
-- Coupon incrementality proxy: was the redeeming household already buying the product?
-- A redemption is "habitual" if the household bought any product covered by that
-- coupon in the 56 days before redeeming (flag defined in mart_coupon_redemptions).
-- Habitual redemptions are likely subsidised purchases that would have happened anyway.
SELECT
  CASE WHEN GROUPING(campaign_type) = 1 THEN 'All' ELSE campaign_type END AS campaign_type,
  COUNT(*) AS redemptions,
  COUNT(*) FILTER (WHERE is_habitual) AS habitual_redemptions,
  ROUND(100.0 * COUNT(*) FILTER (WHERE is_habitual) / COUNT(*), 1) AS habitual_pct
FROM mart_coupon_redemptions
WHERE is_scored
GROUP BY ROLLUP (campaign_type)
ORDER BY GROUPING(campaign_type), campaign_type;
```

| --- | --- | --- | --- |
| Type A | 1,685 | 919 | 54.50 |
| Type B | 300 | 101 | 33.70 |
| Type C | 61 | 16 | 26.20 |
| All | 2,046 | 1,036 | 50.60 |

## Department scorecard: size, reach, discount dependence and own-brand share.

`sql/analysis/03_department_performance.sql`

```sql
-- Department scorecard: size, reach, discount dependence and own-brand share.
-- Business question: "Which departments matter, and how are they sold?"
WITH dept AS (
  SELECT
    p.department,
    SUM(s.sales_value) AS net_sales,
    COUNT(DISTINCT s.household_id) AS buying_households,
    SUM(s.retail_disc + s.coupon_match_disc) AS retailer_discount,
    SUM(s.gross_sales) AS gross_sales,
    SUM(CASE WHEN p.brand = 'Private' THEN s.sales_value END) AS private_sales
  FROM fct_sales s
  JOIN dim_product p ON p.product_id = s.product_id
  WHERE s.is_merch
  GROUP BY p.department
)
SELECT
  department,
  ROUND(net_sales, 2) AS net_sales,
  ROUND(100.0 * net_sales / SUM(net_sales) OVER (), 1) AS sales_share_pct,
  ROUND(100.0 * buying_households / (SELECT COUNT(*) FROM dim_household), 1) AS penetration_pct,
  ROUND(100.0 * retailer_discount / NULLIF(gross_sales, 0), 1) AS discount_rate_pct,
  ROUND(100.0 * COALESCE(private_sales, 0) / NULLIF(net_sales, 0), 1) AS private_label_share_pct
FROM dept
WHERE net_sales > 1000
ORDER BY net_sales DESC;
```
